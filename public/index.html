window.processTextLineup = function() { const teamCode = document.getElementById('adminTeamSelect').value; const pitcherName = document.getElementById('adminPitcherInput').value.trim() || 'Por definir'; const rawText = document.getElementById('adminLineupText').value.trim(); if (!rawText) { alert('⚠️ Por favor ingrese la alineación.'); return; } let parsedBatters = []; // Si el texto pegado contiene guiones entre jugadores (ej: "Roldy Brito - CF Noelvi Marte - RF...") // dividimos la cadena por cada nombre y posición let items = rawText.split(/\s*-\s*/).map(item => item.trim()).filter(item => item.length > 0); if (items.length >= 2) { // Reconstruimos pares: [Nombre, Posición] let currentBatter = items[0]; for (let i = 1; i < items.length; i++) { let parts = items[i].split(/\s+/); let pos = parts[0]; // La posición (CF, RF, LF, DH, C, 2B, 1B, 3B, SS) let nextBatter = parts.slice(1).join(' '); // El nombre del siguiente bateador parsedBatters.push({ pos: `${parsedBatters.length + 1}º`, name: `${currentBatter} - ${pos}` }); currentBatter = nextBatter; } // Agregar el último bateador si quedó pendiente if (currentBatter && parsedBatters.length < 9) { parsedBatters.push({ pos: `${parsedBatters.length + 1}º`, name: currentBatter }); } } else { // Si viene separado por saltos de línea tradicionales const lines = rawText.split('\n').filter(l => l.trim() !== ''); lines.forEach((line, index) => { parsedBatters.push({ pos: `${index + 1}º`, name: line.trim() }); }); } DETAILED_LINEUPS[teamCode] = { pitcher: pitcherName, bullpen: 'Roster activo', batters: parsedBatters }; // Actualiza en cartelera liveGamesData.forEach(game => { if (game.away === teamCode) game.awayPitcher = pitcherName; if (game.home === teamCode) game.homePitcher = pitcherName; }); state.selectedLineupTeam = teamCode; saveState(); renderLineups(); renderGames(); document.getElementById('adminLineupText').value = ''; alert(`✅ Lineup procesado y estructurado del 1º al 9º en filas individuales.`); };
